## PandasTools：把化学数据装进 DataFrame

`PandasTools` 是 RDKit 与 pandas 的桥梁，让你能在 DataFrame 中直接存储和操作分子对象。

### 12.1 从 SDF 加载到 DataFrame

```python
from rdkit.Chem import PandasTools
import pandas as pd

# 从 SDF 加载（保留分子对象）
df = PandasTools.LoadSDF(
    'compounds.sdf',
    smilesName='SMILES',      # SMILES 列名
    molColName='ROMol',       # 分子对象列名
    includeFingerprints=True, # 自动计算指纹列
    removeHs=True,
    strictParsing=True
)

print(f"加载 {len(df)} 个分子")
print(f"列: {list(df.columns)}")
```

### 12.4 批量计算描述符

```python
from rdkit.Chem import Descriptors, rdMolDescriptors

def add_descriptors_to_frame(df, mol_col='ROMol'):
    """为 DataFrame 批量添加常用描述符"""
    mols = df[mol_col]
    
    df['MW'] = mols.apply(Descriptors.MolWt)
    df['LogP'] = mols.apply(Descriptors.MolLogP)
    df['TPSA'] = mols.apply(Descriptors.TPSA)
    df['HBA'] = mols.apply(Descriptors.NumHAcceptors)
    df['HBD'] = mols.apply(Descriptors.NumHDonors)
    df['RotBonds'] = mols.apply(Descriptors.NumRotatableBonds)
    df['AromRings'] = mols.apply(Descriptors.NumAromaticRings)
    df['HeavyAtoms'] = mols.apply(Descriptors.HeavyAtomCount)
    df['Formula'] = mols.apply(rdMolDescriptors.CalcMolFormula)
    
    # QED（类药性综合评分，0-1，越高越好）
    from rdkit.Chem.QED import qed
    df['QED'] = mols.apply(qed)
    
    return df

df_smiles = add_descriptors_to_frame(df_smiles)
print(df_smiles[['ID', 'MW', 'LogP', 'QED']])
```

## 化学空间可视化

### 12.7 化学空间可视化（PCA / t-SNE）

```python
from sklearn.decomposition import PCA
from sklearn.manifold import TSNE
import matplotlib.pyplot as plt
import numpy as np

def visualize_chemical_space(df, mol_col='ROMol', method='pca'):
    """使用 PCA 或 t-SNE 可视化化学空间"""
    from rdkit.Chem import rdFingerprintGenerator
    
    # 生成指纹矩阵
    mfpgen = rdFingerprintGenerator.GetMorganGenerator(
        radius=2, fpSize=2048
    )
    fps = [mfpgen.GetFingerprintAsNumPy(mol) for mol in df[mol_col]]
    fp_matrix = np.array(fps)
    
    # 降维
    if method == 'pca':
        reducer = PCA(n_components=2)
    else:
        reducer = TSNE(n_components=2, random_state=42)
    
    coords = reducer.fit_transform(fp_matrix)
    
    # 绘图
    fig, ax = plt.subplots(figsize=(8, 6))
    scatter = ax.scatter(coords[:, 0], coords[:, 1], 
                         alpha=0.7, s=50)
    ax.set_xlabel(f'{method.upper()} 1')
    ax.set_ylabel(f'{method.upper()} 2')
    ax.set_title(f'化学空间 {method.upper()} 可视化')
    return fig, coords
```

## 多样性筛选

### 12.8 多样性筛选（MaxMin Picker）

从大型化合物库中选出最具多样性（结构差异最大）的子集。

```python
from rdkit.SimDivFilters.rdSimDivPickers import MaxMinPicker

def maxmin_diverse_selection(fps, n_pick=10):
    """使用 MaxMin 算法选取最具多样性的 N 个分子"""
    def dist_func(i, j):
        return 1.0 - DataStructs.TanimotoSimilarity(fps[i], fps[j])
    
    picker = MaxMinPicker()
    picks = picker.LazyPick(dist_func, len(fps), n_pick)
    return list(picks)

# 使用
selected_indices = maxmin_diverse_selection(library_fps, n_pick=10)
diverse_mols = [library_mols[i] for i in selected_indices]
```
